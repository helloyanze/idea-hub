# Idea Hub v2 crontab（用户级，免 sudo）
# 调度器：每 5 分钟唤醒 tick；collect 触发频率由 settings.collect_interval_hours 动态控制
*/5 * * * * cd $HOME/idea-hub && uv run python -m idea_hub.cli tick >> logs/tick.log 2>&1

# 每日备份：SQLite online backup（db.backup API）+ outputs 打包，保留 7 份
0 3 * * * cd $HOME/idea-hub && bash scripts/deploy/backup.sh >> logs/backup.log 2>&1

# Web 服务自启（idea_hub.main:app 同源静态服务 web/dist）
@reboot cd $HOME/idea-hub && nohup uv run uvicorn idea_hub.main:app --host 127.0.0.1 --port 8000 >> logs/server.log 2>&1 &

# 调度器健康监控：每 15 分钟检查 scheduler_last_tick
*/15 * * * * cd $HOME/idea-hub && bash scripts/deploy/healthcheck.sh >> logs/healthcheck.log 2>&1
