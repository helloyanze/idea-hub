# Idea Hub 云端 crontab（用户级，免 sudo）
# mihomo 代理自启
@reboot nohup $HOME/.config/mihomo/mihomo -d $HOME/.config/mihomo -f sub.yaml > $HOME/.config/mihomo/mihomo.log 2>&1 &

# Idea Hub Web 服务自启
@reboot cd $HOME/idea-hub && nohup .venv/bin/uvicorn idea_hub.server:app --host 127.0.0.1 --port 8000 > logs/server.log 2>&1 &

# 每晚 02:00 收集热点并生成 idea（auto_run=0 时跳过）
0 2 * * * cd $HOME/idea-hub && .venv/bin/python -c "from idea_hub import db,models; import sys; c=db.connect('data/idea.db'); sys.exit(0 if models.get_setting(c,'auto_run','1')=='1' else 1)" >/dev/null 2>&1 && export https_proxy=http://127.0.0.1:7890 http_proxy=http://127.0.0.1:7890 && $HOME/.local/bin/uv run python -m idea_hub.cli --db data/idea.db collect >> logs/collect.log 2>&1 && $HOME/.local/bin/hermes chat -q "$(cat $HOME/idea-hub/prompts/generate.txt)" >> logs/generate.log 2>&1 || echo "auto_run 暂停或收集失败，跳过 $(date)" >> logs/auto-run.log

# 每晚 03:00 执行等待队列（auto_run=0 时跳过）
0 3 * * * cd $HOME/idea-hub && .venv/bin/python -c "from idea_hub import db,models; import sys; c=db.connect('data/idea.db'); sys.exit(0 if models.get_setting(c,'auto_run','1')=='1' else 1)" >/dev/null 2>&1 && export https_proxy=http://127.0.0.1:7890 http_proxy=http://127.0.0.1:7890 && $HOME/.local/bin/hermes chat -q "$(cat $HOME/idea-hub/prompts/execute.txt)" >> logs/execute.log 2>&1 || echo "auto_run 暂停，跳过 $(date)" >> logs/auto-run.log

# 每晚 03:30 备份数据库（保留最近 7 份）
30 3 * * * cd $HOME/idea-hub && $HOME/.local/bin/uv run python -c "from idea_hub import db; c=db.connect('data/idea.db'); db.backup_db(c,'backups'); c.close()" >> logs/backup.log 2>&1
