# Idea Hub 云端 crontab（用户级，免 sudo）
# 安装：crontab -l > /tmp/cron.bak 2>/dev/null || true; cat crontab.txt >> /tmp/cron.bak; crontab /tmp/cron.bak
# 注意：把 $HOME 换成实际用户主目录

# 开机自启 Web 服务（免 systemd）
@reboot cd $HOME/idea-hub && nohup .venv/bin/uvicorn idea_hub.server:app --host 127.0.0.1 --port 8000 >> logs/server.log 2>&1 &

# 每晚 02:00 收集热点并生成 idea（Hermes AI 任务）
0 2 * * * cd $HOME/idea-hub && uv run python -m idea_hub.cli --db data/idea.db collect >> logs/collect.log 2>&1 && $HOME/.local/bin/hermes chat -q "$(cat $HOME/idea-hub/prompts/generate.txt)" >> logs/generate.log 2>&1

# 每晚 03:00 执行等待队列（Hermes AI 任务）
0 3 * * * cd $HOME/idea-hub && $HOME/.local/bin/hermes chat -q "$(cat $HOME/idea-hub/prompts/execute.txt)" >> logs/execute.log 2>&1

# 每晚 03:30 备份数据库（保留最近 7 份）
30 3 * * * cd $HOME/idea-hub && uv run python -c "from idea_hub import db; c=db.connect('data/idea.db'); db.backup_db(c,'backups'); c.close()" >> logs/backup.log 2>&1
